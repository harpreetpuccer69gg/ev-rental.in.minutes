<script>
let adminToken = localStorage.getItem('admin_token') || '';
if (adminToken) showDashboard();

function togglePwd() {
  const inp = document.getElementById('adminPass');
  const btn = inp.nextElementSibling;
  if (inp.type === 'password') { inp.type = 'text'; btn.innerText = 'ðŸ™ˆ'; }
  else { inp.type = 'password'; btn.innerText = 'ðŸ‘'; }

async function _doAdminLogin() {
  const email = document.getElementById('adminEmail').value.trim();
  const pass = document.getElementById('adminPass').value;
  if (!email || !pass) return showMsg('loginMsg', 'Please fill all fields', 'error');
  const res = await fetch('/admin/login', {method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify({email, password: pass})});
  const d = await res.json();
  if (d.ok) {
    adminToken = d.token;
    localStorage.setItem('admin_token', adminToken);
    showDashboard();
  } else {
    showMsg('loginMsg', d.msg, 'error');
  }
}

function showDashboard() {
  document.getElementById('loginSection').style.display = 'none';
  document.getElementById('dashSection').style.display = 'block';
  loadPending();
  setInterval(loadPending, 30000); // auto-refresh every 30s
}

async function loadPending() {
  const res = await fetch('/admin/pending', {headers: {'Authorization': 'Bearer ' + adminToken}});
  if (res.status === 401) { doLogout(); return; }
  const d = await res.json();

  const changes = d.changes || [];
  const allVendorRegs = (d.vendors || []);
  const vendors = allVendorRegs.filter(v => v.status === 'pending');
  const pending = changes.filter(c => c.status === 'pending');
  const history = changes.filter(c => c.status !== 'pending');
  const vendorHistory = allVendorRegs.filter(v => v.status !== 'pending');

  document.getElementById('statPending').innerText = pending.length;
  document.getElementById('statVendors').innerText = vendors.length;
  document.getElementById('statTotal').innerText = changes.length;

  // Render pending changes
  const changesArea = document.getElementById('changesArea');
  if (!pending.length) {
    changesArea.innerHTML = '<div style="text-align:center;color:#9e9e9e;padding:30px;">No pending change requests âœ…</div>';
  } else {
    changesArea.innerHTML = pending.map(c => renderChangeCard(c, true)).join('');
  }

  // Render vendor approvals
  const vendorsArea = document.getElementById('vendorsArea');
  if (!vendors.length) {
    vendorsArea.innerHTML = '<div style="text-align:center;color:#9e9e9e;padding:30px;">No pending vendor registrations âœ…</div>';
  } else {
    vendorsArea.innerHTML = vendors.map(v => `
      <div class="vendor-card">
        <div class="vendor-info">
          <div class="vendor-name">${v.vendor_name}</div>
          <div style="color:#757575;font-size:12px;">${v.email} | ${v.phone || 'N/A'}</div>
          <div style="color:#9e9e9e;font-size:11px;margin-top:2px;">Registered: ${new Date(v.created_at).toLocaleString()}</div>
        </div>

          <button class="approve-btn" onclick="reviewVendor('${v.id}','approve')">âœ… Approve</button>
          <button class="reject-btn" onclick="reviewVendor('${v.id}','reject')">âŒ Reject</button>
        </div>
      </div>`).join('');
  }

  // Render history
  const historyArea = document.getElementById('historyArea');
  allHistory = [
    ...history,
    ...vendorHistory.map(v => ({
      id: v.id,
      vendor_name: v.vendor_name,
      type: 'vendor_reg',
      status: v.status,
      submitted_at: v.created_at,
      reviewed_at: null,

function renderChangeCard(c, showActions) {
  const p = c.payload || {};
  const typeClass = c.type === 'new' ? 'type-new' : c.type === 'delete' ? 'type-delete' : 'type-edit';
  const typeLabel = c.type === 'new' ? 'New EV' : c.type === 'delete' ? 'Remove EV' : 'Edit EV';
  const statusClass = c.status === 'approved' ? 'status-approved' : c.status === 'rejected' ? 'status-rejected' : 'status-pending';

  const fields = ['City','Make','Type','Approx Rental/Week','Range (Km)','Security Deposit','Refundable Deposit','Charging/Swap','SPOC','Phone'];
  const rows = fields.filter(f => p[f]).map(f => `<tr><td>${f}</td><td>${p[f]}</td></tr>`).join('');

  return `<div class="change-card">
    <div class="change-header">
      <div>
        <div class="change-vendor">${c.vendor_name}</div>
        <div class="change-meta">Submitted: ${new Date(c.submitted_at).toLocaleString()}</div>
      </div>
      <div style="display:flex;gap:6px;align-items:center;">
        <span class="change-type ${typeClass}">${typeLabel}</span>
        <span class="status-badge ${statusClass}">${c.status}</span>
      </div>
    </div>
    ${p.Image ? `<img src="${p.Image}" style="width:100%;max-height:120px;object-fit:contain;border-radius:8px;border:1px solid #e0e0e0;margin-bottom:10px;"/>` : ''}
    <div class="change-details"><table>${rows}</table></div>
    ${c.review_note ? `<div style="font-size:12px;color:#757575;margin-bottom:10px;">Note: ${c.review_note}</div>` : ''}
    ${showActions ? `
    <div class="action-row">
      <input class="note-inp" id="note_${c.id}" placeholder="Rejection reason (optional)"/>
      <button class="approve-btn" onclick="reviewChange('${c.id}','approved','${(c.vendor_name||'').replace(/'/g,\'')}')">âœ… Approve</button>
      <button class="reject-btn" onclick="reviewChange('${c.id}','rejected','${(c.vendor_name||'').replace(/'/g,\'')}')">âŒ Reject</button>
    </div>` : ''}
  </div>`;
}

async function reviewChange(id, action, vendorName) {
  const note = (document.getElementById('note_' + id) || {}).value || '';
  if (action === 'rejected' && !note) return showToast('Please enter a rejection reason.', 'error');
  const confirmMsg = action === 'approved'
    ? `âœ… Approve change request from "${vendorName}"?`
    : `âŒ Reject change request from "${vendorName}"?`;
  if (!confirm(confirmMsg)) return;
  const btn = event.target;
  btn.disabled = true;
  btn.innerText = action === 'approved' ? 'Approving...' : 'Rejecting...';
  const res = await fetch('/admin/review', {method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify({token: adminToken, change_id: id, action, note})});
  const d = await res.json();
  showToast(d.msg || `Change ${action}`, action === 'approved' ? 'success' : 'error');
  loadPending();
}

async function reviewVendor(id, action) {
  const res = await fetch('/admin/approve-vendor', {method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify({token: adminToken, vendor_id: id, action})});
  const d = await res.json();
  showToast(d.ok ? `Vendor ${action}d successfully` : d.msg, d.ok ? 'success' : 'error');
  loadPending();
}

let allVendors = [];

function switchTab(tab) {
  ['changes','vendors','manage','history'].forEach(t => {
    document.getElementById(t+'Tab').style.display = t === tab ? 'block' : 'none';
  });
  document.querySelectorAll('.tab').forEach((t, i) => {
    t.classList.toggle('active', ['changes','vendors','manage','history'][i] === tab);
  });
  if (tab === 'manage') loadVendors();
}

async function loadVendors() {
  const res = await fetch('/vendors');
  allVendors = await res.json();
  renderVendors(allVendors);
}

function filterVendors() {
  const q = document.getElementById('vendorSearch').value.toLowerCase();
  renderVendors(allVendors.filter(v => (v.Vendor||'').toLowerCase().includes(q) || (v.City||'').toLowerCase().includes(q)));
}

function renderVendors(list) {
  const area = document.getElementById('manageArea');
  if (!list.length) { area.innerHTML = '<div style="text-align:center;color:#9e9e9e;padding:20px;">No vendors found.</div>'; return; }
  const grouped = {};
  list.forEach(v => { if (!grouped[v.Vendor]) grouped[v.Vendor] = []; grouped[v.Vendor].push(v); });
  area.innerHTML = Object.keys(grouped).sort().map(vendor => `
    <div style="background:#f8faff;border-radius:12px;padding:14px;margin-bottom:10px;border:1.5px solid #e3eaff;">

      </div>
      ${grouped[vendor].map(v => `

async function removeVendorEntry(vStr) {
  const v = JSON.parse(vStr);
  if (!confirm(`Remove ${v.Vendor} - ${v.Make} in ${v.City}?`)) return;
  const res = await fetch('/admin/remove-vendor', {method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify({token: adminToken, vendor: v.Vendor, city: v.City, make: v.Make})});
  const d = await res.json();
  alert(d.msg || (d.ok ? 'Removed' : 'Error'));
  if (d.ok) loadVendors();
}

async function submitAddVendor() {
  const vendor = document.getElementById('avVendor').value.trim();
  const city = document.getElementById('avCity').value.trim();
  const make = document.getElementById('avMake').value.trim();
  const rent = document.getElementById('avRent').value.trim();
  if (!vendor || !city || !make || !rent) {
    showAvMsg('Please fill Vendor, City, Make and Rent fields', 'error'); return;
  }
  const btn = document.getElementById('avSubmitBtn');
  btn.innerText = 'Adding...'; btn.disabled = true;
  const payload = {
    Vendor: vendor, City: city, Make: make,
    Type: document.getElementById('avType').value,
    'Approx Rental/Week': rent,
    'Range (Km)': document.getElementById('avRange').value.trim(),
    'Security Deposit': document.getElementById('avDeposit').value.trim(),
    'Refundable Deposit': document.getElementById('avRefund').value.trim(),
    'Charging/Swap': document.getElementById('avCharge').value,
    'Battery Type': document.getElementById('avBattery').value.trim(),
    Status: document.getElementById('avStatus').value,
    SPOC: document.getElementById('avSpoc').value.trim(),
    Phone: document.getElementById('avSpocPhone').value.trim(),
    Image: document.getElementById('avImage').value.trim() || null,
    Email: null, Refundable_Deposit: null
  };
  const res = await fetch('/admin/add-vendor', {method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify({token: adminToken, payload})});
  const d = await res.json();
  btn.innerText = 'Add to Live Listings â†’'; btn.disabled = false;
  if (d.ok) {
    showAvMsg('Added successfully!', 'success');
    setTimeout(() => { closeAddVendor(); loadVendors(); }, 1500);
  } else {
    showAvMsg(d.msg || 'Error adding vendor', 'error');
  }

function showAvMsg(msg, type) {
  const el = document.getElementById('avMsg');
  if (!el) return;
  el.innerText = msg;
  el.style.display = 'block';
  el.style.background = type === 'success' ? '#e8f5e9' : '#ffebee';
  el.style.color = type === 'success' ? '#2e7d32' : '#c62828';
}

let allHistory = [];
let histFilter = 'all';

function filterHistory(status) {
  if (status) {
    histFilter = status;
    ['all','approved','rejected'].forEach(s => {
      const btn = document.getElementById('hf-'+s);
      if (!btn) return;
      btn.style.background = s === status ? '#002d62' : '#fff';
      btn.style.color = s === status ? '#ffd600' : '#374151';
      btn.style.borderColor = s === status ? '#002d62' : '#e8eaf0';
    });
  }
  const q = (document.getElementById('histSearch')||{}).value?.toLowerCase() || '';
  let filtered = allHistory;
  if (histFilter !== 'all') filtered = filtered.filter(c => c.status === histFilter);
  if (q) filtered = filtered.filter(c => (c.vendor_name||'').toLowerCase().includes(q));
  renderHistory(filtered);
}

function renderHistory(list) {
  const area = document.getElementById('historyArea');
  if (!list.length) {
    area.innerHTML = '<div style="text-align:center;color:#9ca3af;padding:40px;font-size:14px;">No records found.</div>';
    return;
  }
  // Group by date
  const grouped = {};
  list.forEach(c => {
    const date = new Date(c.submitted_at).toLocaleDateString('en-IN', {day:'numeric',month:'short',year:'numeric'});
    if (!grouped[date]) grouped[date] = [];
    grouped[date].push(c);
  });
  area.innerHTML = Object.keys(grouped).map(date => `
    <div style="margin-bottom:24px;">
      <div style="font-size:11px;font-weight:700;color:#9ca3af;text-transform:uppercase;letter-spacing:0.8px;margin-bottom:10px;display:flex;align-items:center;gap:8px;">
        <span style="background:#f1f5f9;padding:4px 10px;border-radius:20px;">ðŸ“… ${date}</span>
        <span style="background:#f1f5f9;padding:4px 10px;border-radius:20px;">${grouped[date].length} request${grouped[date].length>1?'s':''}</span>
      </div>
      ${grouped[date].map(c => renderHistoryCard(c)).join('')}
    </div>`).join('');
}

function removeFromHistory(id) {
  if (!confirm('Remove this entry from history and marketplace (if approved)?')) return;
  fetch('/admin/remove-history', {
    method: 'POST',
    headers: {'Content-Type': 'application/json'},
    body: JSON.stringify({token: adminToken, change_id: id})
  }).then(r => r.json()).then(d => {
    if (d.ok) {
      allHistory = allHistory.filter(c => c.id !== id);
      document.getElementById('histTotal').innerText = allHistory.length;
      document.getElementById('histApproved').innerText = allHistory.filter(c=>c.status==='approved').length;
      document.getElementById('histRejected').innerText = allHistory.filter(c=>c.status==='rejected').length;
      filterHistory();
      showToast(d.msg, 'success');
    } else {
      showToast(d.msg || 'Error removing', 'error');
    }
  });
}

function showToast(msg, type='success') {
  const t = document.getElementById('toast');
  t.innerText = (type === 'success' ? 'âœ… ' : 'âŒ ') + msg;
  t.className = 'toast ' + type + ' show';
  setTimeout(() => t.classList.remove('show'), 3000);
}

function showMsg(id, msg, type) {
  const el = document.getElementById(id);
  el.innerText = msg;
  el.className = 'msg ' + type;
}
